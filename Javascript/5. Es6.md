## Goal

Understand the modern JavaScript features introduced in ES6+ and learn why Angular, TypeScript, React, Node.js, and modern JavaScript development heavily depend on them.

---

# What

ES6 (ECMAScript 2015) introduced modern JavaScript features that made code:

```text
Cleaner
Shorter
Readable
Reusable
Maintainable
```

Before ES6:

```javascript
var name = "Deepak";

function greet(name) {
    return "Hello " + name;
}
```

After ES6:

```javascript
const name = "Deepak";

const greet = (name) =>
    `Hello ${name}`;
```

---

# Why

Imagine writing emails.

Old way:

```text
Long
Repeated
Difficult to maintain
```

Modern way:

```text
Short
Clean
Professional
Easy to understand
```

ES6 did the same for JavaScript.

# Hands-On Labs

## Lab 1: let and const

### Before ES6

```javascript
var technology = "Angular";
```

### ES6

```javascript
let technology = "Angular";

const company = "TechLambda";

console.log(technology);
console.log(company);
```

---

### Difference

```text
let
    Value can change

const
    Value cannot change
```

---

## Lab 2: Template Literals

### Old Way

```javascript
let name = "Deepak";

console.log(
    "Hello " + name
);
```

### ES6 Way

```javascript
let name = "Deepak";

console.log(
    `Hello ${name}`
);
```

Output:

```text
Hello Deepak
```

---

## Lab 3: Arrow Functions

### Old Way

```javascript
function add(a, b) {
    return a + b;
}
```

### ES6 Way

```javascript
const add =
    (a, b) => a + b;

console.log(add(10, 20));
```

Output:

```text
30
```

---

## Lab 4: Default Parameters

```javascript
function greet(
    name = "Guest"
) {

    console.log(
        `Hello ${name}`
    );

}

greet();

greet("Deepak");
```

Output:

```text
Hello Guest
Hello Deepak
```

---

## Lab 5: Rest Parameters

```javascript
function total(
    ...numbers
) {

    let sum = 0;

    for(let num of numbers){
        sum += num;
    }

    return sum;
}

console.log(
    total(10,20,30,40)
);
```

Output:

```text
100
```

---

## Lab 6: Spread Operator

```javascript
let frontend = [
    "HTML",
    "CSS"
];

let fullStack = [
    ...frontend,
    "JavaScript",
    "Angular"
];

console.log(fullStack);
```

Output:

```text
HTML
CSS
JavaScript
Angular
```

---

## Lab 7: Array Destructuring

```javascript
let technologies = [
    "JavaScript",
    "TypeScript",
    "Angular"
];

let [
    first,
    second,
    third
] = technologies;

console.log(first);
console.log(second);
console.log(third);
```

Output:

```text
JavaScript
TypeScript
Angular
```

---

## Lab 8: Object Destructuring

```javascript
let trainer = {

    name: "Deepak",

    technology: "Angular"

};

let {
    name,
    technology
} = trainer;

console.log(name);
console.log(technology);
```

---

## Lab 9: Classes

```javascript
class Trainer {

    constructor(
        name,
        technology
    ) {

        this.name =
            name;

        this.technology =
            technology;
    }

    teach() {

        console.log(
            `${this.name}
            teaches
            ${this.technology}`
        );

    }
}

let trainer1 =
    new Trainer(
        "Deepak",
        "Angular"
    );

trainer1.teach();
```

---

## Lab 10: Promises

```javascript
let coursePromise =
    new Promise(
        (resolve) => {

            resolve(
                "Angular Course Loaded"
            );

        }
    );

coursePromise.then(
    result => {

        console.log(result);

    }
);
```

Output:

```text
Angular Course Loaded
```

---

## Lab 11: Async Await

```javascript
async function loadCourse() {

    let course =
        await Promise.resolve(
            "Angular Loaded"
        );

    console.log(course);
}

loadCourse();
```

Output:

```text
Angular Loaded
```
